━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  VaultSync (sync.py)
  └─ Orchestrates all vault operations
     • check_connection()    - Test connectivity
     • push_report()         - Send findings to vault
     • pull_templates()      - Get templates
     • search_notes()        - Search vault
     • full_sync()           - Bidirectional sync

  ObsidianVaultClient (vault_api.py)
  └─ Communicates with Obsidian via REST API
     • create_note()         - Create/overwrite note
     • read_note()           - Read note content
     • append_to_note()      - Add to existing note
     • is_reachable()        - Test connection

  NoteBuilder (note_builder.py)
  └─ Structures findings into Obsidian notes
     • set_title()           - Set note title
     • set_type()            - Set note type
     • add_tag()             - Add tags
     • add_section()         - Add content section
     • add_callout()         - Add highlight box
     • build()               - Generate markdown
